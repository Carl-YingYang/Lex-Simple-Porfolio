    60→/**
    61→ * Hard-shadow press helper ("sink into paper") — pareho ng helper ng
    62→ * ProfileScreen. Para lang sa focal element na may 2px hard shadow
    63→ * (dito: I-save button ng rename modal).
    64→ */
    65→function makeSink(colors: ThemeColors) {
    66→    return (pressed: boolean, level: 'sm' | 'md' | 'lg'): ViewStyle => {
    67→        if (!pressed) return getNeoShadow(colors, level);
    68→        const offset = SHADOW[level].ios.shadowOffset;
    69→        return {
    70→            elevation: 0,
    71→            transform: [
    72→                { translateX: offset.width },
    73→                { translateY: offset.height },
    74→            ],
    75→        };
    76→    };
    77→}

   142→const getHistoryImages = (
   143→    item: ScanHistoryItem
   144→): string[] => {
   145→    const candidates = item.pageUris?.length
   146→        ? item.pageUris
   147→        : item.images?.length
   148→          ? item.images
   149→          : item.uri
   150→            ? [item.uri]
   151→            : [];
   152→
   153→    return candidates.filter(
   154→        (value): value is string =>
   155→            typeof value === 'string' &&
   156→            value.trim().length > 0
   157→    );
   158→};
   159→
   160→const getItemDate = (value: string): string => {
   161→    if (!value) {
   162→        return 'Walang petsa';
   163→    }
   164→
   165→    return value.split(',')[0]?.trim() || value;
   166→};

   168→const isGenericTitle = (value: string): boolean => {
   169→    const normalized = value.trim().toLocaleLowerCase();
   170→
   171→    return (
   172→        normalized === 'camera scan' ||
   173→        normalized === 'gallery upload' ||
   174→        normalized === 'document file' ||
   175→        normalized === 'document scan' ||
   176→        normalized.startsWith('document scan (') ||
   177→        normalized.includes('pages)')
   178→        || /^dokumento\s+\d+\s*·\s*pahina/i.test(normalized)
   179→    );
   180→};
